6pub type ChallengeTimeMilliseconds = u64;
7pub type ChallengeId = String;
8pub type ChallengePercentage = u8;
9pub type ChallengePerformance = (ChallengeId, ChallengePercentage, ChallengeTimeMilliseconds);

11#[derive(Serialize, Deserialize, Debug, Clone, PartialEq, Eq, Hash, Default)]
12pub struct PerformanceRecord {
13    pub game_path_id: String,
14    pub profile_name: String,
15    pub challenges_performance: Vec<ChallengePerformance>,
16    pub total_challenges: usize,
17    pub performance_percentage: u8,
18    pub date: DateTime<Utc>,
19}

21impl PerformanceRecord {
22    pub fn new(
23        game_path_id: String,
24        user_profile_name: String,
25        challenges_performance: Vec<ChallengePerformance>,
26        total_challenges: usize,
27        date: DateTime<Utc>,
28    ) -> Self {
29        let solved_challenges = challenges_performance.len();
30        let performance_percentage =
31            ((solved_challenges as f64 / total_challenges as f64) * 100.0) as u8;
32        PerformanceRecord {
33            game_path_id,
34            profile_name: user_profile_name,
35            challenges_performance,
36            total_challenges,
37            performance_percentage,
38            date,
39        }
40    }
41
42    pub fn new_from_history(
43        game_path_id: String,
44        user_profile_name: String,
45        total_challenges: usize,
46        challenge_history: ChallengeHistory,
47    ) -> Self {
48        let mut best_performance: HashMap<ChallengeId, ChallengePerformance> = HashMap::new();
49
50        for challenge in challenge_history.challenges {
51            let id = challenge.challenge_config.id.clone();
52            let performance = challenge.performance(&challenge.challenge_result) as u8;
53            let elapsed_time = challenge
54                .elapsed_time()
55                .map(|d| d.num_milliseconds() as ChallengeTimeMilliseconds)
56                .unwrap_or(0);
57
58            best_performance
59                .entry(id.clone())
60                .and_modify(|(_id, p, t)| {
61                    if performance > *p {
62                        // New best performance - update both performance and time
63                        *p = performance;
64                        *t = elapsed_time;
65                    } else if performance == *p && elapsed_time < *t {
66                        // Same performance but faster time - update time only
67                        *t = elapsed_time;
68                    }
69                })
70                .or_insert((id, performance, elapsed_time));
71        }
72
73        let mut challenges_performance: Vec<ChallengePerformance> =
74            best_performance.values().cloned().collect();
75
76        // Sort by performance in descending order (highest performance first)
77        challenges_performance.sort_by_key(|b| std::cmp::Reverse(b.1));
78
79        // Sort by elapsed time in ascending order (lowest time first)
80        challenges_performance.sort_by_key(|a| a.2);
81
82        // Calculate overall performance percentage based on best performances
83        let performance_percentage = if !challenges_performance.is_empty() {
84            // Use the best performance for each unique challenge
85            let total_performance: u32 = challenges_performance
86                .iter()
87                .map(|(_, perf, _)| *perf as u32)
88                .sum();
89
90            // Average of best performances
91            (total_performance / challenges_performance.len() as u32) as u8
92        } else {
93            0
94        };
95
96        PerformanceRecord {
97            game_path_id,
98            profile_name: user_profile_name,
99            challenges_performance,
100            total_challenges,
101            performance_percentage,
102            date: Utc::now(),
103        }
104    }
105}

107impl Timed for PerformanceRecord {
108    fn start(&mut self) {}
109    fn update_end_time(&mut self) {}
110    fn elapsed_time(&self) -> Option<Duration> {
111        let mut elapsed_time = Duration::zero();
112        for (_, _, time) in &self.challenges_performance {
113            elapsed_time += Duration::milliseconds(*time as i64);
114        }
115        Some(elapsed_time)
116    }
117    fn start_time(&self) -> Option<DateTime<Utc>> {
118        Some(self.date)
119    }
120    fn end_time(&self) -> Option<DateTime<Utc>> {
121        let end_time = self.date + self.elapsed_time().unwrap_or_default();
122        Some(end_time)
123    }
124}

126impl Ord for PerformanceRecord {
127    fn cmp(&self, other: &Self) -> std::cmp::Ordering {
128        match other
129            .performance_percentage
130            .cmp(&self.performance_percentage)
131        {
132            std::cmp::Ordering::Equal => self
133                .elapsed_time()
134                .and_then(|self_elapsed| {
135                    other
136                        .elapsed_time()
137                        .map(|other_elapsed| self_elapsed.cmp(&other_elapsed))
138                })
139                .map(|time_ord| time_ord.then_with(|| other.date.cmp(&self.date)))
140                .unwrap_or(std::cmp::Ordering::Equal),
141            other => other,
142        }
143    }
144}
145
146impl PartialOrd for PerformanceRecord {
147    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
148        Some(self.cmp(other))
149    }
