1use super::{
2    EventType,
3    event::{Event, EventTrait},
4};
5use std::collections::HashMap;
6use std::sync::{Arc, Mutex};
7
8type EventHandler = Arc<dyn Fn(Event) + Send + Sync>;
9
10#[derive(Default, Clone)]
11pub struct EventBus {
12    listeners: Arc<Mutex<HashMap<EventType, Vec<EventHandler>>>>,
13}
14
15impl EventBus {
16    pub fn new() -> Self {
17        Self::default()
18    }
19
20    pub fn subscribe<F>(&self, event_type: EventType, callback: F)
21    where
22        F: Fn(Event) + Send + Sync + 'static,
23    {
24        let mut listeners = self
25            .listeners
26            .lock()
27            .unwrap_or_else(|poisoned| poisoned.into_inner());
28        listeners
29            .entry(event_type)
30            .or_default()
31            .push(Arc::new(callback));
32    }
33
34    /// Returns the number of event types with registered listeners.
35    pub fn listener_count(&self) -> usize {
36        self.listeners
37            .lock()
38            .unwrap_or_else(|poisoned| poisoned.into_inner())
39            .len()
40    }
41
42    pub fn publish(&self, event: Event) {
43        let listeners = self
44            .listeners
45            .lock()
46            .unwrap_or_else(|poisoned| poisoned.into_inner());
47        if let Some(handlers) = listeners.get(&event.get_type()) {
48            for handler in handlers {
49                handler(event.clone());
50            }
51        }
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::super::{GameEvent, event::Event};
58    use super::*;
59    use std::sync::Arc;
60    use std::sync::atomic::{AtomicUsize, Ordering};
61
62    #[test]
63    fn test_event_bus() {
64        let event_bus = EventBus::new();
65        let counter = Arc::new(AtomicUsize::new(0));
66        let counter_clone = counter.clone();
67        event_bus.subscribe(EventType::Game, move |event| {
68            if let Event::Game(GameEvent::Started) = event {
69                counter_clone.fetch_add(1, Ordering::SeqCst);
70            }
71        });
72
73        event_bus.publish(Event::Game(GameEvent::Started));
74        assert_eq!(counter.load(Ordering::SeqCst), 1);
75    }
