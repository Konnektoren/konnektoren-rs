32/// A backend-agnostic key, decoupled from any specific terminal or input crate.
33///
34/// Native (crossterm) input is mapped to this type internally; other frontends
35/// (e.g. a web-based terminal) can map their own key events to it and drive
36/// [`App`] via [`App::handle_key`] without this crate depending on their input crate.
37#[derive(Debug, Clone, Copy, PartialEq, Eq)]
38pub enum Key {
39    Char(char),
40    Up,
41    Down,
42    Left,
43    Right,
44    Tab,
45    BackTab,
46    Enter,
47    Esc,
48    PageUp,
49    PageDown,
50    Home,
51}
52
53#[cfg(feature = "crossterm")]
54impl Key {
55    fn from_crossterm(code: KeyCode) -> Option<Self> {
56        Some(match code {
57            KeyCode::Char(c) => Key::Char(c),
58            KeyCode::Up => Key::Up,
59            KeyCode::Down => Key::Down,
60            KeyCode::Left => Key::Left,
61            KeyCode::Right => Key::Right,
62            KeyCode::Tab => Key::Tab,
63            KeyCode::BackTab => Key::BackTab,
64            KeyCode::Enter => Key::Enter,
65            KeyCode::Esc => Key::Esc,
66            KeyCode::PageUp => Key::PageUp,
67            KeyCode::PageDown => Key::PageDown,
68            KeyCode::Home => Key::Home,
69            _ => return None,
70        })
71    }
72}

149        terminal.clear().map_err(Error::Io)?;

266    pub fn open_challenge(&mut self, index: usize) -> Result<()> {
267        let challenge_id = self
268            .current_game_path()
269            .and_then(|path| path.challenges.get(index))
270            .map(|challenge| challenge.id.clone())
271            .ok_or_else(|| Error::State(format!("challenge index not found: {index}")))?;
272
273        let mut challenge = self
274            .session
275            .game_state
276            .game
277            .create_challenge(&challenge_id)
278            .map_err(Error::Game)?;
279        challenge.start();
280
281        self.session.game_state.challenge = challenge;
282        self.session.game_state.current_challenge_index = index;
283        self.session.game_state.current_task_index = 0;
284        self.selected_challenge_index = index;
285        Ok(())
286    }

291            .game
292            .game_paths
293            .get(self.session.game_state.current_game_path)
294    }

296    /// Handle a backend-agnostic key. See [`Key`].
297    pub fn handle_key(&mut self, key: Key) -> Result<()> {
298        match key {
299            Key::Char('q') | Key::Esc => self.exit(),
300            Key::Char('c') => self.show_challenge_page(),
301            Key::Char('g') => self.show_challenge_list(),
302            Key::Char('i') => self.show_challenge_info(),
303            Key::Char('m') => self.toggle_map(),
304            Key::Up | Key::Char('k') if self.page == AppPage::Info => {
305                self.scroll_info_up(1);
306            }
307            Key::Down | Key::Char('j') if self.page == AppPage::Info => {
308                self.scroll_info_down(1);
309            }
310            Key::PageUp if self.page == AppPage::Info => {
311                self.scroll_info_up(10);
312            }
313            Key::PageDown if self.page == AppPage::Info => {
314                self.scroll_info_down(10);
315            }
316            Key::Home if self.page == AppPage::Info => {
317                self.reset_info_scroll();
318            }
319            Key::Up | Key::Char('k') if self.page == AppPage::Challenges => {
320                self.select_previous_challenge_in_list();
321            }
322            Key::Down | Key::Char('j') if self.page == AppPage::Challenges => {
323                self.select_next_challenge_in_list();
324            }
325            Key::Enter if self.page == AppPage::Challenges => self.open_selected_challenge()?,
326            Key::Left | Key::Char('h') => self.previous_question(),
327            Key::Right | Key::Char('l') => self.next_question(),
328            Key::Tab => self.next_challenge(),
329            Key::BackTab => self.previous_challenge(),
330            Key::Char('0') => self.solve_option(0)?,
331            Key::Char('1') => self.solve_option(1)?,
332            Key::Char('2') => self.solve_option(2)?,
333            Key::Char('3') => self.solve_option(3)?,
334            Key::Char('4') => self.solve_option(4)?,
335            Key::Char('5') => self.solve_option(5)?,
336            Key::Char('6') => self.solve_option(6)?,
337            Key::Char('7') => self.solve_option(7)?,
338            Key::Char('8') => self.solve_option(8)?,
339            Key::Char('9') => self.solve_option(9)?,
340            _ => {}
341        }
342        Ok(())
343    }
344
345    #[cfg(feature = "crossterm")]
346    fn handle_key_event(&mut self, key_event: KeyEvent) -> Result<()> {
347        let Some(key) = Key::from_crossterm(key_event.code) else {
348            return Ok(());
349        };
350        self.handle_key(key)
351    }
352
353    #[cfg(feature = "crossterm")]
354    fn handle_events(&mut self) -> Result<()> {
355        match event::read().map_err(Error::Io)? {
356            Event::Key(key_event) if key_event.kind == KeyEventKind::Press => {
357                if let Err(e) = self.handle_key_event(key_event) {
358                    tracing::error!("Error handling key event: {}", e);
359                }
360            }
361            _ => {}
362        };
363        Ok(())
364    }

367impl Widget for &App {
368    fn render(self, area: Rect, buf: &mut Buffer) {
369        let username_display = self
370            .username
371            .as_ref()
372            .map(|u| format!(" {}: {} ", self.t("User"), u))
373            .unwrap_or_else(|| self.title.clone());
374        let header = if let Some(language) = &self.language {
375            format!("{} {}: {} ", username_display, self.t("Language"), language)
376        } else {
377            username_display
378        };
379
380        let instructions = Line::from(vec![
381            format!(" {} ", self.t("Previous")).into(),
382            "<Left>".blue().bold(),
383            format!(" {} ", self.t("Next")).into(),
384            "<Right>".blue().bold(),
385            format!(" {} ", self.t("Map")).into(),
386            "<M>".blue().bold(),
387            format!(" {} ", self.t("List")).into(),
388            "<G>".blue().bold(),
389            format!(" {} ", self.t("Play")).into(),
390            "<C>".blue().bold(),
391            format!(" {} ", self.t("Info")).into(),
392            "<I>".blue().bold(),
393            format!(" {} ", self.t("Quit")).into(),
394            "<Q> ".blue().bold(),
395        ]);
396
397        let block = Block::default()
398            .title(Line::from(header).bold().centered())
399            .title_bottom(instructions.centered())
400            .borders(Borders::ALL)
401            .border_set(border::THICK);
402
403        Paragraph::new(":")
404            .centered()
405            .block(block)
406            .render(area, buf);
407
408        let inner_area = area.inner(Margin {
409            horizontal: 1,
410            vertical: 1,
411        });
412
413        let vertical = Layout::vertical([Constraint::Length(1), Constraint::Min(0)]);
414        let [page_tabs_area, content_area] = vertical.areas(inner_area);
415        PageTabs::new(
416            self.page.tab(),
417            &self.i18n,
418            self.selected_language().as_ref(),
419        )
420        .render(page_tabs_area, buf);
421
422        let Some(game_path) = self.current_game_path() else {
423            Paragraph::new(self.t("No game path")).render(content_area, buf);
424            return;
425        };
426
427        match self.page {
428            AppPage::Map => {
429                MapWidget::new(
430                    game_path,
431                    self.session.game_state.current_challenge_index,
432                    &self.i18n,
433                    self.selected_language().as_ref(),
434                )
435                .render(content_area, buf);
436            }
437            AppPage::Challenges => {
438                let mut state = ratatui::widgets::ListState::default();
439                ChallengeList::new(
440                    game_path,
441                    self.session.game_state.current_challenge_index,
442                    self.selected_challenge_index,
443                    &self.i18n,
444                    self.selected_language().as_ref(),
445                )
446                .render(content_area, buf, &mut state);
447            }
448            AppPage::Info => {
449                if let Some(config) = game_path.challenges.get(self.selected_challenge_index) {
450                    let created = self.session.game_state.game.create_challenge(&config.id);
451                    let active = (self.selected_challenge_index
452                        == self.session.game_state.current_challenge_index)
453                        .then_some(&self.session.game_state.challenge);
454                    ChallengeDebugInfo::new(
455                        config,
456                        self.selected_challenge_index,
457                        created.as_ref(),
458                        active,
459                        self.info_scroll,
460                        &self.i18n,
461                        self.selected_language().as_ref(),
462                    )
463                    .render(content_area, buf);
464                } else {
465                    Paragraph::new(self.t("No challenge selected")).render(content_area, buf);
466                }
467            }
468            AppPage::Challenge => {
469                let vertical = Layout::vertical([Constraint::Length(1), Constraint::Min(0)]);
470                let [tab_area, challenge_area] = vertical.areas(content_area);
471                ChallengeTabs::new(game_path, self.session.game_state.current_challenge_index)
472                    .render(tab_area, buf);
473
474                ChallengeWidget {
475                    challenge: &self.session.game_state.challenge,
476                    show_help: true,
477                    current_question: self.session.game_state.current_task_index,
478                    i18n: &self.i18n,
479                    language: self.selected_language().as_ref(),
480                }
481                .render(challenge_area, buf);
482            }
483        }
484    }
485}
