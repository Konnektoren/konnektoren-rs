14pub struct ChallengeDebugInfo<'a> {
15    config: &'a ChallengeConfig,
16    selected_index: usize,
17    created: std::result::Result<&'a Challenge, &'a GameError>,
18    active: Option<&'a Challenge>,
19    scroll: u16,
20    i18n: &'a I18nConfig,
21    language: Option<&'a Language>,

24impl<'a> ChallengeDebugInfo<'a> {
25    pub fn new(
26        config: &'a ChallengeConfig,
27        selected_index: usize,
28        created: std::result::Result<&'a Challenge, &'a GameError>,
29        active: Option<&'a Challenge>,
30        scroll: u16,
31        i18n: &'a I18nConfig,
32        language: Option<&'a Language>,
33    ) -> Self {
34        Self {
35            config,
36            selected_index,
37            created,
38            active,
39            scroll,
40            i18n,
41            language,
42        }
43    }
44
45    fn t(&self, key: &str) -> String {
46        self.language.map_or_else(
47            || self.i18n.t(key),
48            |language| self.i18n.t_with_lang(key, language),
49        )
50    }
51}

53impl Widget for ChallengeDebugInfo<'_> {
54    fn render(self, area: Rect, buf: &mut Buffer) {
55        Paragraph::new(self.lines())
56            .block(Block::bordered().title(format!(" {} ", self.t("Challenge Debug Info"))))
57            .scroll((self.scroll, 0))
58            .wrap(Wrap { trim: false })
59            .render(area, buf);
60    }
61}

63impl ChallengeDebugInfo<'_> {
64    fn lines(&self) -> Text<'static> {
65        let mut lines = vec![
66            Line::from(self.t("Config").bold()),
67            Line::from(format!("index: {}", self.selected_index)),
68            Line::from(format!("id: {}", self.config.id)),
69            Line::from(format!("name: {}", self.config.name)),
70            Line::from(format!("description: {}", self.config.description)),
71            Line::from(format!("challenge asset id: {}", self.config.challenge)),
72            Line::from(format!(
73                "variant: {}",
74                self.config
75                    .variant
76                    .as_ref()
77                    .map(ToString::to_string)
78                    .unwrap_or_else(|| "-".to_string())
79            )),
80            Line::from(format!("tasks: {}", self.config.tasks)),
81            Line::from(format!("unlock_points: {}", self.config.unlock_points)),
82            Line::from(format!("position: {:?}", self.config.position)),
83            Line::from(format!(
84                "icon: {}",
85                self.config.icon.as_deref().unwrap_or("-")
86            )),
87            Line::from(""),
88            Line::from(self.t("Asset validation").bold()),
89        ];
90
91        match self.created {
92            Ok(challenge) => {
93                lines.push(Line::from(format!("status: {}", self.t("Ok")).green()));
94                lines.push(Line::from(format!(
95                    "challenge type id: {}",
96                    challenge.challenge_type.id()
97                )));
98                lines.push(Line::from(format!(
99                    "challenge type name: {}",
100                    challenge.challenge_type.name()
101                )));
102                push_yaml(
103                    &mut lines,
104                    self.t("Challenge type yaml"),
105                    &challenge.challenge_type,
106                );
107                push_yaml(
108                    &mut lines,
109                    self.t("Initial result yaml"),
110                    &challenge.challenge_result,
111                );
112            }
113            Err(err) => {
114                lines.push(Line::from(
115                    format!("status: {}", self.t("Failed")).red().bold(),
116                ));
117                lines.push(Line::from(format!("error: {err}")));
118            }
119        }
120
121        lines.push(Line::from(""));
122        lines.push(Line::from(self.t("Active runtime").bold()));
123        if let Some(challenge) = self.active {
124            lines.push(Line::from(format!("active id: {}", challenge.get_id())));
125            lines.push(Line::from(format!("solved: {}", challenge.solved())));
126            lines.push(Line::from(format!(
127                "performance: {}",
128                challenge.performance(&challenge.challenge_result)
129            )));
130            lines.push(Line::from(format!(
131                "start_time: {:?}",
132                challenge.start_time()
133            )));
134            lines.push(Line::from(format!("end_time: {:?}", challenge.end_time())));
135            lines.push(Line::from(format!(
136                "elapsed_seconds: {}",
137                challenge
138                    .elapsed_time()
139                    .map(|elapsed| elapsed.num_seconds().to_string())
140                    .unwrap_or_else(|| "-".to_string())
141            )));
142            push_yaml(
143                &mut lines,
144                self.t("Config yaml"),
145                &challenge.challenge_config,
146            );
147            push_yaml(
148                &mut lines,
149                self.t("Result yaml"),
150                &challenge.challenge_result,
151            );
152        } else {
153            lines.push(Line::from(self.t("Not the active challenge")));
154        }
155
156        Text::from(lines)
157    }
158}

160fn push_yaml<T>(lines: &mut Vec<Line<'static>>, title: String, value: &T)
161where
162    T: serde::Serialize,
163{
164    lines.push(Line::from(""));
165    lines.push(Line::from(title.to_string().bold()));
166    for line in to_yaml(value).lines() {
167        lines.push(Line::from(line.to_string()));
168    }
169}
170
171fn to_yaml<T>(value: &T) -> String
172where
173    T: serde::Serialize,
174{
175    serde_yaml::to_string(value)
176        .map(|yaml| yaml.trim().to_string())
177        .unwrap_or_else(|err| format!("serialization failed: {err}"))
178}
