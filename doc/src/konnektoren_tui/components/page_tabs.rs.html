1use konnektoren_platform::i18n::{I18nConfig, Language};
2use ratatui::{
3    buffer::Buffer,
4    layout::Rect,
5    style::{Color, Modifier, Style},
6    widgets::{Tabs, Widget},
7};
8
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum PageTab {
11    Challenge,
12    Challenges,
13    Map,
14    Info,
15}
16
17impl PageTab {
18    pub const fn index(self) -> usize {
19        match self {
20            Self::Challenge => 0,
21            Self::Challenges => 1,
22            Self::Map => 2,
23            Self::Info => 3,
24        }
25    }
26}
27
28pub struct PageTabs {
29    selected: PageTab,
30    labels: [String; 4],
31}
32
33impl PageTabs {
34    pub fn new(selected: PageTab, i18n: &I18nConfig, language: Option<&Language>) -> Self {
35        Self {
36            selected,
37            labels: [
38                t(i18n, language, "Challenge"),
39                t(i18n, language, "Challenges"),
40                t(i18n, language, "Map"),
41                t(i18n, language, "Info"),
42            ],
43        }
44    }
45}
46
47impl Widget for PageTabs {
48    fn render(self, area: Rect, buf: &mut Buffer) {
49        Tabs::new(self.labels)
50            .highlight_style(Style::new().bg(Color::Blue).add_modifier(Modifier::BOLD))
51            .select(self.selected.index())
52            .padding(" ", " ")
53            .divider(" | ")
54            .render(area, buf);
55    }
56}
57
58fn t(i18n: &I18nConfig, language: Option<&Language>, key: &str) -> String {
59    language.map_or_else(|| i18n.t(key), |language| i18n.t_with_lang(key, language))
60}
