3pub trait Translation {
4    fn t(&self, key: &str) -> String;
5    fn t_with_lang(&self, key: &str, lang: &Language) -> String;
6
7    /// Returns `Some(translation)` if found in the default language, `None` if missing.
8    fn try_t(&self, key: &str) -> Option<String> {
9        let result = self.t(key);
10        if result == key { None } else { Some(result) }
11    }
12
13    /// Returns `Some(translation)` if found in `lang`, `None` if missing.
14    fn try_t_with_lang(&self, key: &str, lang: &Language) -> Option<String> {
15        let result = self.t_with_lang(key, lang);
16        if result == key { None } else { Some(result) }
17    }
18}

25    /// Always translates: returns a prefixed string different from the key.
26    struct DummyTranslation;
27    impl Translation for DummyTranslation {
28        fn t(&self, key: &str) -> String {
29            format!("t:{}", key)
30        }
31        fn t_with_lang(&self, key: &str, lang: &Language) -> String {
32            format!("t:{}:{}", key, lang.code())
33        }
34    }
35
36    /// Simulates a missing translation: returns the key unchanged (fallback behaviour).
37    struct FallbackTranslation;
38    impl Translation for FallbackTranslation {
39        fn t(&self, key: &str) -> String {
40            key.to_string()
41        }
42        fn t_with_lang(&self, key: &str, _lang: &Language) -> String {
43            key.to_string()
44        }
45    }

50        assert_eq!(dummy.t("hello"), "t:hello");
51        let lang = Language::from("de");
