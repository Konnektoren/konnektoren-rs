3/// I18n configuration declared in a manifest YAML.
4///
5/// Tells loaders where translation files live, which languages are supported,
6/// and which one to use when no preference is expressed.
7///
8/// ```yaml
9/// i18n:
10///   path: assets/i18n
11///   default_language: en
12///   languages: [de, en, es, pl, tr, uk, ar, zh]
13/// ```
14#[derive(Debug, Clone, PartialEq, Deserialize, Serialize)]
15pub struct I18nManifest {
16    /// Path to the directory containing per-language JSON/YAML files,
17    /// relative to the crate's asset root.
18    #[serde(default = "default_i18n_path")]
19    pub path: String,
20
21    /// BCP-47 language code for the fallback / default language.
22    #[serde(default = "default_language")]
23    pub default_language: String,
24
25    /// Ordered list of BCP-47 language codes the deployment supports.
26    /// An empty list means "accept whatever files are found in `path`".
27    #[serde(default)]
28    pub languages: Vec<String>,
29}
30
31impl Default for I18nManifest {
32    fn default() -> Self {
33        Self {
34            path: default_i18n_path(),
35            default_language: default_language(),
36            languages: vec![],
37        }
38    }
39}
40
41fn default_i18n_path() -> String {
42    "i18n".to_string()
43}
44
45fn default_language() -> String {
46    "en".to_string()
47}

49#[cfg(test)]
50mod tests {
51    use super::*;
52
53    const FULL_YAML: &str = r#"
54path: i18n
55default_language: de
56languages:
57  - de
58  - en
59  - es
60  - pl
61  - tr
62  - uk
63  - ar
64  - zh
65"#;
66
67    #[test]
68    fn test_deserializes_full() {
69        let i: I18nManifest = serde_yaml::from_str(FULL_YAML).unwrap();
70        assert_eq!(i.path, "i18n");
71        assert_eq!(i.default_language, "de");
72        assert_eq!(i.languages.len(), 8);
73        assert!(i.languages.contains(&"uk".to_string()));
74    }
75
76    #[test]
77    fn test_defaults_when_empty() {
78        let i: I18nManifest = serde_yaml::from_str("{}").unwrap();
79        assert_eq!(i.path, "i18n");
80        assert_eq!(i.default_language, "en");
81        assert!(i.languages.is_empty());
82    }
83
84    #[test]
85    fn test_default_impl() {
86        let i = I18nManifest::default();
87        assert_eq!(i.path, "i18n");
88        assert_eq!(i.default_language, "en");
89    }
90
91    #[test]
92    fn test_partial_override() {
93        let yaml = "default_language: tr\n";
94        let i: I18nManifest = serde_yaml::from_str(yaml).unwrap();
95        assert_eq!(i.default_language, "tr");
96        assert_eq!(i.path, "i18n"); // still default
97    }
